## CRIAR POLITICA
$policy = @'
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": "sqs:GetQueueAttributes",
      "Resource": "arn:aws:sqs:us-east-1:608541039538:fiap-evaluation-analytics-queue"
    }
  ]
}
'@

$policyPath = Join-Path (Get-Location) 'keda-sqs-policy.json'

[IO.File]::WriteAllText(
    $policyPath,
    $policy,
    [Text.UTF8Encoding]::new($false)
)

$policyArn = aws iam list-policies `
    --scope Local `
    --query "Policies[?PolicyName=='ToggleMasterKedaSqsReadPolicy'].Arn | [0]" `
    --output text

if ($LASTEXITCODE -ne 0 -or [string]::IsNullOrWhiteSpace($policyArn) -or $policyArn -eq 'None') {
    throw 'Não foi possível localizar a política existente.'
}

Write-Host "Política encontrada: $policyArn"

kubectl create namespace keda --dry-run=client -o yaml | kubectl apply -f -

eksctl create iamserviceaccount `
    --cluster fiap-techchallenge `
    --region us-east-1 `
    --namespace keda `
    --name keda-operator `
    --role-name ToggleMasterKedaRole `
    --attach-policy-arn $policyArn `
    --override-existing-serviceaccounts `
    --approve

kubectl annotate serviceaccount keda-operator `
    -n keda `
    eks.amazonaws.com/sts-regional-endpoints=true `
    --overwrite
